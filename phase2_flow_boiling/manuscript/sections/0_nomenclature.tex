\section*{Nomenclature}
\addcontentsline{toc}{section}{Nomenclature}

\noindent
\begin{minipage}[t]{0.48\textwidth}
\textit{Latin symbols}\\[2pt]
\begin{tabular}{@{}l l@{}}
$Bo$        & boiling number, $\qpp/(G\hfg)$ \\
$\Dh$       & hydraulic diameter, m \\
$G$         & mass flux, kg\,m$^{-2}$\,s$^{-1}$ \\
$\hfg$      & latent heat of vaporization, J\,kg$^{-1}$ \\
$k$         & coverage interval half-width, $\times\sigma$ \\
$K$         & deep-ensemble size \\
$P$         & pressure, Pa \\
$P_r$       & reduced pressure, $P/P_{\mathrm{crit}}$ \\
$\qpp$      & heat flux, W\,m$^{-2}$ \\
$\qonb$     & ONB heat flux, W\,m$^{-2}$ \\
$Re$        & Reynolds number \\
$R^2$       & coefficient of determination \\
$T_w$       & wall temperature, K \\
$\Tsat$     & saturation temperature, K \\
$We$        & Weber number \\
$x^{*}$     & normalized axial coordinate \\
$\mathbf{z}_s,\mathbf{z}_f$ & surface/flow latent vectors \\
\end{tabular}
\end{minipage}
\hfill
\begin{minipage}[t]{0.48\textwidth}
\textit{Greek symbols}\\[2pt]
\begin{tabular}{@{}l l@{}}
$\dTonb$    & ONB wall superheat, $T_w-\Tsat$, K \\
$\dThsu$    & saturated-pool Hsu superheat, K \\
$\dTsub$    & inlet/local subcooling, K \\
$\rhol,\rhov$ & liquid/vapour density, kg\,m$^{-3}$ \\
$\sigma$    & surface tension, N\,m$^{-1}$ \\
$\theta$    & static contact angle, deg \\
\end{tabular}
