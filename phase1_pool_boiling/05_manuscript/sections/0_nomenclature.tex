\section*{Nomenclature}
\addcontentsline{toc}{section}{Nomenclature}

\noindent\textit{Roman symbols}
\begin{tabbing}
\hspace{2.4cm}\=\hspace{10cm}\=\kill
$\mathrm{Bi}$      \> Biot number, $h_{nc}L/k_s$                          \> [---] \\
$\mathrm{Bo}$      \> Bond number, $(\rho_{\ell}-\rho_v)gL^2/\sigma$        \> [---] \\
$C_1$              \> Hsu wettability coefficient, $(1+\cos\theta)/\sin\theta$  \> [---] \\
$C_2$              \> Hsu wettability coefficient, $(1+\cos\theta)/\sin^{2}\theta$ \> [---] \\
$c_{p,\ell}$       \> liquid isobaric specific heat                       \> [\si{J/(kg.K)}] \\
$d_z$              \> surface latent-vector dimension                     \> [---] \\
$D_{\mathrm{Hsu}}$ \> Hsu nucleation discriminant                         \> [---] \\
$F(\theta)$        \> Basu contact-angle modifier                         \> [---] \\
$g$                \> gravitational acceleration                          \> [\si{m/s^2}] \\
$\mathrm{Gr}_L$    \> Grashof number based on $L$                         \> [---] \\
$h_{fg}$           \> latent heat of vaporization                         \> [\si{J/kg}] \\
$h_{nc}$           \> natural-convection heat-transfer coefficient        \> [\si{W/(m^2.K)}] \\
$\mathrm{Ja}$      \> Jacob number, $\rho_{\ell}c_{p,\ell}\Delta T/(\rho_v h_{fg})$ \> [---] \\
$K$                \> Zuber CHF correction constant                       \> [---] \\
$k_{\ell}$         \> liquid thermal conductivity                         \> [\si{W/(m.K)}] \\
$k_s$              \> solid (heater) thermal conductivity                 \> [\si{W/(m.K)}] \\
$L_c$              \> capillary length, $\sqrt{\sigma/(g\Delta\rho)}$     \> [m] \\
$L_w$              \> heater-wall thickness                               \> [m] \\
$n$                \> number of ONB evaluation points                     \> [---] \\
$N_s$              \> active nucleation-site density                      \> [\si{1/cm^2}] \\
$\mathrm{Nu}_L$    \> Nusselt number based on $L$                         \> [---] \\
$P$                \> system pressure                                     \> [Pa] \\
$p_r$              \> reduced pressure, $P/P_{\mathrm{crit}}$             \> [---] \\
$\mathrm{Pr}$      \> Prandtl number, $\mu_{\ell}c_{p,\ell}/k_{\ell}$     \> [---] \\
$q''$              \> wall heat flux                                      \> [\si{W/m^2}] \\
$q_{\mathrm{ref}}$ \> Zuber-type heat-flux scale, $h_{fg}\rho_v^{1/2}[\sigma g\Delta\rho]^{1/4}$ \> [\si{W/m^2}] \\
$\mathrm{Ra}_L$    \> Rayleigh number based on $L$                        \> [---] \\
$R^{2}$            \> coefficient of determination                        \> [---] \\
$R_a$              \> arithmetic-mean surface roughness                   \> [\si{\micro m}] \\
$r_c$              \> active cavity radius                                \> [\si{\micro m}] \\
$T_{\mathrm{sat}}$ \> saturation temperature at $P$                       \> [K] \\
$T_w$              \> heater-wall (heated-side) temperature               \> [K] \\
$T_{\infty}$       \> bulk-liquid temperature                             \> [K] \\
$z$                \> coordinate normal to the heater wall                \> [m] \\
$z_s$              \> surface latent vector from FiLM encoder             \> [---] \\
\end{tabbing}

\noindent\textit{Greek symbols}
\begin{tabbing}
\hspace{2.4cm}\=\hspace{10cm}\=\kill
$\alpha_{\ell}$    \> liquid thermal diffusivity                          \> [\si{m^2/s}] \\
$\beta_{\ell}$     \> liquid volumetric thermal expansion coefficient     \> [\si{1/K}] \\
$\delta_t$         \> thermal-boundary-layer thickness                    \> [m] \\
$\Delta T_{\mathrm{ONB}}$ \> wall superheat at ONB, $T_w-T_{\mathrm{sat}}$ \> [K] \\
$\Delta T_{\mathrm{ref}}$ \> latent-heat temperature scale, $h_{fg}/c_{p,\ell}$ \> [K] \\
$\Delta T_{\mathrm{sub}}$ \> subcooling, $T_{\mathrm{sat}}-T_{\infty}$    \> [K] \\
$\Delta\rho$       \> density difference $\rho_{\ell}-\rho_v$             \> [\si{kg/m^3}] \\
$\varepsilon$      \> Hsu soft-constraint threshold                       \> [---] \\
$\theta$           \> static contact angle                                \> [\si{\degree}] \\
$\mu_{\ell}$       \> liquid dynamic viscosity                            \> [\si{Pa.s}] \\
$\nu_{\ell}$       \> liquid kinematic viscosity                          \> [\si{m^2/s}] \\
$\rho$             \> Spearman rank-correlation coefficient (Sec.~\ref{subsec:physics}) \> [---] \\
$\rho_{\ell}$      \> liquid density                                      \> [\si{kg/m^3}] \\
$\rho_v$           \> vapor density                                       \> [\si{kg/m^3}] \\
$\sigma$           \> surface tension                                     \> [\si{N/m}] \\
$\sigma_{\mathrm{epi}}$  \> epistemic standard deviation                  \> [K] \\
$\sigma_{\mathrm{ale}}$  \> aleatoric standard deviation                  \> [K] \\
\end{tabbing}
